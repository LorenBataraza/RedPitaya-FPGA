% El pool de batches de la sesión del osciloscopio, en un instante cualquiera.
% No es un batch: son n_batches EventBatch (entre 2 y 8; 8 con los valores por
% defecto) que se crean una vez, al armar la sesión, y circulan siempre los
% mismos: free_q -> el lector, que lo llena -> full_q -> el escritor, que lo
% guarda -> free_q. Cada uno lleva su nombre (B0 a B7) para que se vea que son
% objetos y no copias. El lazo de C escribe adentro del que tiene el lector,
% desde batch.n; sin uno libre, el lector llama a read_batch(None, …) y la fuente
% decide qué pasa con los eventos.
% Fuentes: software/API/osciloscope_store/{events.py (EventBatch, make_pool,
% batches_for_budget), session.py, reader.py (_loop_lotes), storage.py
% (ChunkWriter._loop)}. El texto que acompaña está en
% docs/API/adquisicion_de_eventos.md §4; el epígrafe, en pool_batches.txt.
% Estilo: docs/figuras/estilo_tikz.tex.
%
% Compilar:   pdflatex pool_batches.tex
%             pdftocairo -svg pool_batches.pdf pool_batches.svg
%

% \batch{nodo}{posición}{nombre}{fracción llena}: un EventBatch. Blanco = libre;
% la parte azul son las filas ya escritas.
\newcommand{\batch}[4]{%
  \node[draw=black, line width=0.6pt, fill=white, minimum width=0.95cm,
        minimum height=0.6cm, inner sep=0pt] (#1) at (#2) {};
  \ifdim #4pt>0pt
    \fill[azul!40] (#1.south west) rectangle ($(#1.south west)!#4!(#1.south east)+(0,0.6)$);
  \fi
  \draw[line width=0.6pt] (#1.south west) rectangle (#1.north east);
  \node[font=\scriptsize\ttfamily\bfseries] at (#1) {#3};}
