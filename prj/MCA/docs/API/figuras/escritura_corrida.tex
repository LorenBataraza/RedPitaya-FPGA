% Lo que la API escribe cuando mide: el directorio de una corrida. Los dos
% escritores van primero a un .tmp y después hacen os.replace, así que todo
% .npz o .json visible está completo. Debajo, el camino físico de un write().
% Fuentes: software/API/osciloscope_store/storage.py (ChunkWriter._write),
% session.py (AcquisitionSession, run_summary.json). El texto que acompaña está en
% docs/API/arquitectura_software.md §3. Estilo: docs/figuras/estilo_tikz.tex.
%
% Compilar:   pdflatex escritura_corrida.tex
%             pdftocairo -svg escritura_corrida.pdf escritura_corrida.svg
%
\documentclass[border=0pt,tikz]{standalone}
\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}

\tikzset{
  hilo/.style    = {bloque=azul!30, text width=2.6cm, minimum height=0.9cm},
  archivo/.style = {draw=black, line width=0.5pt, fill=white, font=\scriptsize\ttfamily,
                    minimum width=2.9cm, minimum height=0.38cm, inner sep=1pt},
  tmp/.style     = {archivo, dashed, draw=black!60, text=black!65},
  fis/.style     = {bloque=grisr, text width=2.3cm, minimum height=0.9cm},
}

\begin{document}

\draw[flujo] (pc) -- node[sen, above] {writeback} (sd);
