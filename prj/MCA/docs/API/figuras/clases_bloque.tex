% La forma de un bloque de libmca: lo que comparten todos los handles, dibujado
% una sola vez. Las figuras de los bloques (clases_libmca) y de los lazos
% (clases_lotes) no lo repiten.
%   arriba     quién la llama (el binding, por ctypes; un programa en C) y el
%              handle genérico <bloque>_h, con mca_bloque como primer miembro (la
%              herencia a la manera de C); a la derecha, lo que es común a la
%              interfaz: los códigos, la versión y las tablas de registros;
%   abajo      la única puerta a la PL: mca_bus con el backend devmem_bus, y la
%              PL; a la izquierda, mca_mem, la región de DDR que sólo usa el ring.
% Los backends por callbacks y por fichero (bus_callback.c, el _open_file de
% bus_mmap.c) no se dibujan: sirven a las pruebas (diseno_de_clases.md §5.1).
% Fuentes: software/libmca/include/{mca_api.h,mca_bus.h,mca_mem.h},
% software/libmca/src/{bloque.h,bus.h,bus.c,devmem.h,bus_devmem.c,bus_mmap.c,mem.c}.
% El texto que acompaña está en docs/API/libmca.md §1 y §2; el epígrafe, en
% clases_bloque.txt. Estilo: docs/figuras/estilo_tikz.tex y clases_estilo.tex.
%
% Compilar:   pdflatex clases_bloque.tex
%             pdftocairo -svg clases_bloque.pdf clases_bloque.svg
%

\node[ctipo=6.6cm, anchor=north] (h) at (0,0) {
  \clase{<bloque>_h}{include/<bloque>.h · src/<bloque>.c · struct opaco}
  \nodepart{two}
  b : mca_bloque \es{primer miembro}\\
  lo propio del bloque: la geometría, la DDR, …
  \nodepart{three}
  \gr{ciclo de vida}
  <bloque>_open(\&h, phys, size, …) · <bloque>_close(h)\\
  <bloque>_open_bus(\&h, bus, propio, …)\\
  <bloque>_last_error(h)\\
  \gr{el mapa, de {\ttfamily <bloque>_regs.h}}
  <bloque>_reg_count() · <bloque>_reg_info(i)\\
  <bloque>_r32(h, off, \&v) · <bloque>_w32(h, off, v)\\
  \gr{un campo}
  <bloque>_get_<campo>(h, \&v) · <bloque>_set_<campo>(h, v)\\
  \gr{una operación}
  int <bloque>_<op>(h, …, \&salida) -> MCA_OK | MCA_E*
};
