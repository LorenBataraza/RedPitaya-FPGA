% El MCA desde Python: el binding de libmca y el camino remoto. Lo que hace
% cada función está en la figura de la capa de C (clases_libmca): acá no se
% repite, porque el binding tiene una función por cada una de C, con el mismo
% nombre. Tres columnas:
%   izquierda  la red: MCARemote, el proxy, que hereda del binding sin abrir
%              libmca y manda cada función entera por la op `call`, y el
%              servidor, que la corre del otro lado sobre su handle;
%   centro     MCA, el handle de Python: guarda el mca_h* y la geometría;
%   derecha    las funciones mca_* y el despacho de @_api según h._via: a
%              libmca por ctypes, a la referencia en Python o a la red.
% Los sustitutos sin placa (FakeMCA y el backend por callbacks) no se dibujan:
% están en diseno_de_clases.md §9.1.
% Fuentes: software/API/mca.py (class MCA, mca_open, @_api),
% software/API/_despacho.py, software/API/_libmca.py,
% software/API/mca_remote.py (MCARemote), software/API/ref_py/mca2.py,
% software/app/mca_server.py. El texto que acompaña está en
% docs/API/diseno_de_clases.md §4 y §8. Estilo: docs/figuras/estilo_tikz.tex y
% clases_estilo.tex.
%
% Compilar:   pdflatex clases_mca.tex
%             pdftocairo -svg clases_mca.pdf clases_mca.svg
%

  \clase{MCA}{API/mca.py · binding de libmca · 0x4070_0000 · slot 7}
  \nodepart{two}
  _c : mca_h*\\
  caps · h_aw · h2_aw · psd_aw · _has_2d\\
  _via = 'c' · libmca_version

  close() · __enter__() · __exit__()\\
  \gr{el resto: uno por función de libmca}
  configure() · counters() · spectrum() · …
};

% ============================================= las funciones de Python ===
\node[modulo=5.6cm, anchor=north] (fun) at (\xC,0) {
  {\bfseries\fontsize{7.8}{9.2}\selectfont mca_*(h, …)}\\[0.8pt]
  {\fontsize{5.6}{6.6}\selectfont API/mca.py · la superficie pública · @_api}\\[3pt]
  una por función de libmca:\\
  \hspace*{1em}mismo nombre, mismos argumentos\\[2pt]
  según h._via:\\
  \hspace*{1em}'c'      -> libmca por ctypes, con h._c\\
  \hspace*{1em}'py'     -> la homónima de ref_py\\
  \hspace*{1em}'remoto' -> h.call(nombre, …)\\[2pt]
  mca_open(impl) -> MCA | la referencia\\
  API_REGISTRADA: el registro de @_api
};
\draw[usa] ([yshift=-1.0cm]mca.north east) -- ([yshift=-1.0cm]mca.north east -| fun.west)
  node[rol, midway, above] {self};

\node[modulo=4.6cm, anchor=north, fill=rojo!30] (lib) at (\xC,-4.9) {
  {\bfseries\fontsize{7.8}{9.2}\selectfont libmca}\\[0.8pt]
  {\fontsize{5.6}{6.6}\selectfont figura clases_libmca}\\[3pt]
  mca_h* · mca_*(h, …, \&out)\\
  devuelven MCA_OK o un MCA_E*
};
\draw[usa] (fun.south) -- (lib.north) node[rol, midway, right] {'c' · ctypes};

\node[modulo=4.6cm, anchor=north] (ref) at (\xC,-7.6) {
  {\bfseries\fontsize{7.8}{9.2}\selectfont API/ref_py/mca2.py}\\[0.8pt]
  {\fontsize{5.6}{6.6}\selectfont la referencia en Python · sobre mmap}\\[3pt]
  la misma superficie, sin C\\
  ref_py/mca.py: la del MCA1
};
\draw[usa] (fun.east) -- ++(0.55,0) |- (ref.east) node[rol, pos=0.25, right] {'py'};

% ===================================================== la red: el proxy ===
\node[clase=5.6cm, anchor=north] (rem) at (\xA,0) {
  \clase{MCARemote}{API/mca_remote.py · proxy de la op call · en la PC}
  \nodepart{two}
  _canal · _lock · info\\

  \nodepart{three}
  connect(host, port) \es{classmethod}\\
  call(nombre, *a, **k)\\

};
% hereda del binding, sin llamar a su constructor (abriría libmca)
\draw[hereda] ([yshift=-1.0cm]rem.north east) -- ([yshift=-1.0cm]rem.north east -| mca.west);

\node[modulo=4.6cm, anchor=north] (srv) at (\xA,-7.0) {
  {\bfseries\fontsize{7.8}{9.2}\selectfont ServidorMCA}\\[0.8pt]

};
\draw[usa] (rem.south) -- (srv.north)
  node[rol, pos=0.35, right] {_canal} node[rol, pos=0.7, right] {TCP · op call};
% el servidor abre un MCA: entra por el binding
\draw[usa] (srv.east) -| (mca.south) node[rol, pos=0.25, above] {h};
