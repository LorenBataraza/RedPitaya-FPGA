% Por qué hace falta Channel. El socket TCP es un flujo de bytes sin límites
% entre mensajes (al medio: tres respuestas seguidas, placa -> PC). Arriba, lo
% que trae cada recv(): pedazos de cualquier largo, y algunos mezclan partes
% distintas (el final de una cabecera y el principio del payload, o el final del
% payload y la cabecera siguiente). Abajo, los cortes de Channel.recv():
% readline() hasta el '\n' y read(nbytes) para el payload, y lo que devuelve.
% Lo rayado es lo que se leyó de más y queda en el buffer de lectura, que dura
% lo que la conexión: por eso Channel es un objeto y no dos funciones.
% Fuentes: software/API/mca_net.py (Channel.recv, _read_exact; el
% BufferedReader de sock.makefile('rb')). El texto que acompaña está en
% docs/APP/protocolo.md, "La trama"; el epígrafe, en trama_flujo.txt. Estilo:
% docs/figuras/estilo_tikz.tex.
%
% Compilar:   pdflatex trama_flujo.tex
%             pdftocairo -svg trama_flujo.pdf trama_flujo.svg
%
\documentclass[border=0pt,tikz]{standalone}
\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage[none]{hyphenat}
\input{../../figuras/estilo_tikz.tex}
\usetikzlibrary{patterns}

\tikzset{
  seg/.style    = {draw=black, line width=0.5pt},
  txt/.style    = {font=\tiny\ttfamily, inner sep=0pt},
  fila/.style   = {anchor=east, font=\scriptsize, align=right, inner sep=0pt},
  rec/.style    = {line width=0.7pt, black!70},
  cruza/.style  = {line width=0.9pt, rojo!80!black},
  cha/.style    = {line width=0.7pt, azul!80!black},
}
\newcommand{\LF}{\textbackslash n}
% un tramo de la cinta: x0, x1, relleno
\newcommand{\tramo}[3]{\filldraw[seg, fill=#3] (#1,-0.23) rectangle (#2,0.23);}
% un corte de recv(), arriba: x0, x1, estilo
\newcommand{\recv}[3]{\draw[#3] (#1+0.03,0.33) -- (#1+0.03,0.55) -- (#2-0.03,0.55) -- (#2-0.03,0.33);
  \node[txt, font=\scriptsize\ttfamily, #3, above=2pt] at ({(#1+#2)/2},0.55) {recv()};}
% un corte de Channel, abajo: x0, x1, y, rótulo
\newcommand{\cort}[4]{\draw[cha] (#1+0.03,#3+0.2) -- (#1+0.03,#3) -- (#2-0.03,#3) -- (#2-0.03,#3+0.2);
  \node[txt, font=\scriptsize\ttfamily, azul!60!black, below=2pt, align=center] at ({(#1+#2)/2},#3) {#4};}
% corte de "sin escala" sobre la cinta, en x
\newcommand{\corte}[1]{%
  \fill[white] (#1-0.07,-0.23) -- (#1+0.03,0.23) -- (#1+0.15,0.23) -- (#1+0.05,-0.23) -- cycle;
  \draw[line width=0.5pt] (#1-0.07,-0.23) -- (#1+0.03,0.23) (#1+0.05,-0.23) -- (#1+0.15,0.23);}

% ===================================================== arriba: recv() ===
\recv{0}{3.65}{cruza}
\recv{3.65}{7.6}{cruza}
\recv{7.6}{10.3}{rec}
\node[font=\small, black!70] at (11.05,0.62) {…};
\recv{11.8}{14.3}{cruza}
\recv{14.3}{16.15}{rec}

% ============================================ abajo: Channel.recv() ===
% lo que lee cada paso
\cort{0}{2.75}{-0.45}{readline()}
\cort{2.75}{6.6}{-0.45}{readline()}
\cort{6.6}{13.4}{-0.45}{read(65536)}
\cort{13.4}{16.15}{-0.45}{readline()}
% lo que devuelve
\cort{0}{2.75}{-1.2}{(cabecera, None)}
\cort{2.75}{13.4}{-1.2}{(cabecera, payload)}
\cort{13.4}{16.15}{-1.2}{(cabecera, None)}
