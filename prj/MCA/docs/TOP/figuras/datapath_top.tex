% Ruteo del datapath del top en integración, dibujado como crossbar: UNA cadena
% de acondicionado por canal, cuatro nodos utilizables sobre ella (A..D) y un
% dsp_tap_mux 4:1 por consumidor. Donde hay punto hay camino; el número es el
% valor de ROUTE_* que lo elige; el punto con anillo es el valor de reset.
%
% Las marcas numeradas son las filas de la tabla de
% docs/TOP/register_map_top.md §Ruteo del datapath: si se cambia una marca, se
% cambia esa tabla. Antes la generaba figuras_datapath_top.py; ahora es TikZ,
% con el estilo de docs/figuras/estilo_tikz.tex.
% Fuentes: modulos/osc/rtl/rp_scope_mt_com.sv, modulos/comun/rtl/dsp_tap_mux.sv.
%
% Compilar:   pdflatex datapath_top.tex
%             pdftocairo -svg datapath_top.pdf datapath_top.svg
%
\documentclass[border=0pt,tikz]{standalone}
\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage[none]{hyphenat}
\input{../../figuras/estilo_tikz.tex}

\tikzset{
  cad/.style    = {datos, text width=2.4cm, minimum height=0.75cm},
  letra/.style  = {circle, draw=black, line width=0.6pt, fill=white,
                   font=\scriptsize\bfseries, inner sep=0pt, minimum size=4.4mm},
  riel/.style   = {->, line width=0.9pt},
  col/.style    = {line width=0.6pt, black!45},
  chip/.style   = {draw=black!60, fill=white, rounded corners=1.5pt,
                   font=\scriptsize\ttfamily, inner sep=2pt},
  tapn/.style   = {font=\scriptsize\ttfamily, inner sep=1pt, anchor=north west},
}
% punto de cruce; con * es el valor de reset (anillo)
\newcommand{\cruza}[2]{\fill (#1) circle (1.6pt);
  \node[tapn] at ($(#1)+(0.12,-0.08)$) {#2};}
\newcommand{\reset}[2]{\draw[line width=0.7pt, fill=white] (#1) circle (3.6pt);
  \fill (#1) circle (1.6pt);
  \node[tapn, font=\scriptsize\ttfamily\bfseries] at ($(#1)+(0.2,-0.1)$) {#2};}
% trapecio del mux, ancho arriba
\newcommand{\muxv}[2]{% #1 = centro, #2 = nombre
  \draw[line width=0.6pt, fill=grisr] ($(#1)+(-0.95,0.28)$) -- ($(#1)+(0.95,0.28)$)
    -- ($(#1)+(0.65,-0.28)$) -- ($(#1)+(-0.65,-0.28)$) -- cycle;
  \node[font=\tiny\ttfamily\bfseries] at (#1) {dsp\_tap\_mux};
  \coordinate (#2) at (#1);}

\begin{document}
\begin{tikzpicture}[fondo]

% ===================================================== la cadena, en vertical
\node[cad] (adc) at (1.2,0)    {\texttt{\bfseries ADC}};
\node[cad] (cal) at (1.2,-1.8) {\texttt{\bfseries rp\_scope\_calib}};
\node[mudo, text width=2.4cm, minimum height=0.75cm] (flt) at (1.2,-3.0) {\texttt{\bfseries osc\_filter}};
\coordinate (mx) at (1.2,-4.0);
\draw[line width=0.6pt, fill=azul!35] ($(mx)+(-0.55,0.22)$) -- ($(mx)+(0.55,0.22)$)
  -- ($(mx)+(0.35,-0.22)$) -- ($(mx)+(-0.35,-0.22)$) -- cycle;
\node[font=\tiny\ttfamily\bfseries] at (mx) {byp};
\node[cad] (dec) at (1.2,-5.4) {\texttt{\bfseries rp\_decim}};
\node[cad] (dly) at (1.2,-7.0) {\texttt{\bfseries rp\_delay}};
\node[ddr, text width=2.4cm, minimum height=0.75cm] (axi) at (1.2,-9.6)
  {\texttt{\bfseries rp\_axi\_sm}\\[-1pt]{\scriptsize$\to$ DDR (AXI HP)}};

% los cuatro nodos
\coordinate (nA) at (1.2,-0.9);
\coordinate (nB) at (1.2,-4.65);

\coordinate (nD) at (1.2,-7.8);

\draw[flujo] (adc.south) -- (cal.north);
\draw[flujo] (cal.south) -- (flt.north);
\draw[flujo] (flt.south) -- ($(mx)+(0,0.22)$);
\coordinate (tb) at ($(cal.south)!0.5!(flt.north)$);
\fill (tb) circle (1.3pt);
\draw[flujo] (tb) -- ++(-1.6,0) |- ($(mx)+(-0.45,0.4)$) -- ($(mx)+(-0.45,0.22)$);

% ============================================================== los rieles
\def\xfin{15.2}
\foreach \n/\l/\s in {nA/A/adc\_calib\_in, nB/B/adc\_dec\_in, nC/C/adc\_dly\_in, nD/D/adc\_bram\_in} {
  \draw[riel] (\n) -- (\xfin,0 |- \n);
  \node[chip, anchor=south west] at ($(\n)+(2.0,0.08)$) {\s};
}
\foreach \n/\l in {nA/A, nB/B, nC/C, nD/D}
  \node[letra] at (\n) {\l};

% ================================================== columnas y cruces
\def\xo{8.2}\def\xm{10.9}\def\xc{13.6}
\foreach \x/\t in {\xo/OSC, \xm/MULTITRIGGER, \xc/MCA} {
  \node[font=\scriptsize\bfseries] at (\x,-0.3) {\t};
  \draw[col] (\x,-0.55) -- (\x,-8.72);
}
% A: los tres consumidores pueden tomar el crudo
\cruza{\xo,-0.9}{1} \cruza{\xm,-0.9}{1} \cruza{\xc,-0.9}{1}
% B: el MCA, de reset
\cruza{\xo,-4.65}{2} \cruza{\xm,-4.65}{2} \reset{\xc,-4.65}{0 = 2}
% C: sólo el multitrigger, y sólo como nativo
\reset{\xm,-6.2}{0}
% D: el OSC, de reset
\reset{\xo,-7.8}{0 = 3} \cruza{\xm,-7.8}{3} \cruza{\xc,-7.8}{3}

% ======================================================= muxes y consumidores
\muxv{\xo,-9.0}{mo} \muxv{\xm,-9.0}{mm} \muxv{\xc,-9.0}{mc}
\node[bloque=azul!35, text width=2.45cm, minimum height=0.7cm] (co) at (\xo,-10.25) {\textbf{OSC}\\ slot 1};
\node[trig, text width=2.45cm, minimum height=0.7cm] (cm) at (\xm,-10.25) {\textbf{MULTITRIGGER}\\ slot 3};
\node[captura, text width=2.45cm, minimum height=0.7cm] (cc) at (\xc,-10.25) {\textbf{MCA}\\ slot 7};
\foreach \m/\c in {mo/co, mm/cm, mc/cc}
  \draw[flujo] ($(\m)+(0,-0.28)$) -- (\c.north);
% el veto es la única señal que cruza de un consumidor a otro
\draw[flujo] (cm.south) -- ++(0,-0.45) -| node[sen, above, pos=0.3] {veto} (cc.south);
