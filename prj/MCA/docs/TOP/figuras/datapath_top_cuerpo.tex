% dsp_tap_mux 4:1 por consumidor. No es una matriz de conmutación: cada

% la letra del nodo en vez de dibujar un cable a cada uno, para que no parezca
% una grilla de cruces.
%
% Las marcas numeradas (sólo con \ringfalse) son las filas de la tabla de
% docs/TOP/register_map_top.md §Ruteo del datapath: si se cambia una marca, se

% mux 2:1 chico (el lazo digital y el bypass del ecualizador)
\newcommand{\muxdos}[2]{% #1 = nombre, #2 = centro
  \draw[line width=0.6pt, fill=grisr] ($(#2)+(-0.22,0.42)$) -- ($(#2)+(0.22,0.25)$)
    -- ($(#2)+(0.22,-0.25)$) -- ($(#2)+(-0.22,-0.42)$) -- cycle;
  \coordinate (#1) at (#2);
  \coordinate (#1-a) at ($(#2)+(-0.22,0.2)$);
  \coordinate (#1-b) at ($(#2)+(-0.22,-0.2)$);
  \coordinate (#1-o) at ($(#2)+(0.22,0)$);
  \coordinate (#1-s) at ($(#2)+(0,-0.33)$);
  \coordinate (#1-n) at ($(#2)+(0,0.33)$);}
% un dsp_tap_mux: trapecio con las cuatro entradas arriba, rotuladas con el
% código de tap y la letra del nodo; #1 = nombre, #2 = centro, #3..#6 = nodos
% de las entradas 0..3
\newcommand{\tapmux}[6]{%
  \muxdecv{#1}{(#2)}{2.6}
  \node[font=\tiny\ttfamily\bfseries] at (#1) {dsp\_tap\_mux};
  \foreach \k/\n [count=\i from 0] in {0/#3, 1/#4, 2/#5, 3/#6} {
    \coordinate (#1-e\k) at ($(#1)+(-0.9+0.6*\i,0.45)$);
    \draw[line width=0.6pt] (#1-e\k) -- ++(0,0.32);
    \node[letra] at ($(#1-e\k)+(0,0.55)$) {\n};
    \node[font=\tiny\ttfamily, anchor=north] at ($(#1-e\k)+(0,-0.02)$) {\k};
  }}
