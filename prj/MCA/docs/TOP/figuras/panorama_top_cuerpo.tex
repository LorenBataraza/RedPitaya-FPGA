% RELOJES. El relleno sigue siendo la función; el reloj va en una etiqueta en
% el borde del bloque, y sólo donde NO es adc_clk (o donde podría confundirse,
% como axi_master, que vive en el PS pero corre en adc_clk). Los cruces que
% necesitan sincronizador llevan la marca CDC: fclk[0] y adc_clk tienen la misma
% frecuencia nominal pero salen de osciladores distintos (plesiócronos), y el PS
% cruza a sus relojes internos en GP0 y HP. Fuente: las instancias de
% rtl/mine/tops/mca_red_pitaya_top.sv y rtl/red_pitaya_ps.sv (ACLK de GP0 y HP).
\tikzset{
  reloj/.style = {draw=black!75, fill=white, line width=0.4pt, rounded corners=1.2pt,
                  font=\fontsize{5.4}{6}\selectfont\ttfamily, inner ysep=1.1pt,
                  inner xsep=1.3pt,
                  path picture={
                    \draw[line width=0.35pt]
                      ([xshift=1.25mm]path picture bounding box.west) circle (0.75mm)
                      ([xshift=1.25mm]path picture bounding box.west) -- ++(0,0.52mm)
                      ([xshift=1.25mm]path picture bounding box.west) -- ++(0.4mm,0);}},
  cdc/.style   = {fill=black, text=white, rounded corners=1pt, inner sep=1.1pt,
                  font=\fontsize{5.4}{6}\selectfont\bfseries},
}
