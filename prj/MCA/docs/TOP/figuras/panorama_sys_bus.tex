\node[etapa=white, text width=5.4cm, align=left, font=\scriptsize, anchor=north] (ncdc) at (11.3,-8.35)
  {\textbf{Dentro de cada \texttt{sys\_bus\_cdc}} (handshake por \emph{toggle}):\\[2pt]
   ida: \texttt{ctrl\_do} cruza por 3 FF a \texttt{reg\_do};\\
   vuelta: \texttt{reg\_done} cruza por 3 FF y da \texttt{ack}.\\[2pt]
   \texttt{addr}, \texttt{wdata} y \texttt{rdata} no se sincronizan:\\
   el protocolo los mantiene quietos.\\[2pt]
   PLL sin enganchar: \texttt{rdata = 0xDEADBEEF}.\\
   El \texttt{err} del esclavo no vuelve.};

   \texttt{ack} de latencia fija, que no mire\\ la dirección: sin \texttt{ack} el slot\\ queda muerto.\\[2pt]

\node[etapa=white, text width=5.6cm, align=left, font=\scriptsize, anchor=north] (nps) at (2.95,-8.35)
