% El contrato del bus de sistema como máquina de estados, vista desde el puerto
% del esclavo (adc_clk). Las transiciones en negro las hace el maestro (el
% sys_bus_cdc); la verde es la obligación del esclavo. No hay un estado de
% falla: si el ack no llega, el esclavo se queda en ESPERA, porque del lado de
% adc_clk no hay timeout. El de axi4_slave libera al ARM, no al slot (nota roja).
% Fuentes: rtl/sys_bus_cdc.sv (reg_write/reg_read se limpian sólo con ack;
% addr y wdata se cargan al arrancar y quedan quietos), el en_sr de
% integration_cfg.sv, mt_cfg.sv, osc_cfg.sv y mca_top.sv, rtl/axi4_slave.sv
% (timeout de 32 ciclos). Documentado en docs/TOP/bus_sistema_redpitaya.md §7.1
% y §8. Estilo: docs/figuras/estilo_tikz.tex.
%
% Compilar:   pdflatex contrato_sys_bus.tex
%             pdftocairo -svg contrato_sys_bus.pdf contrato_sys_bus.svg
%
\documentclass[border=0pt,tikz]{standalone}
\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage[none]{hyphenat}
\input{../../figuras/estilo_tikz.tex}

\tikzset{
  estado/.style   = {draw=black, line width=0.8pt, rounded corners=9pt, fill=white,
                     align=center, text width=2.55cm, minimum height=1.55cm,
                     inner sep=3pt, font=\scriptsize},
  maestro/.style  = {->, line width=0.9pt},
  esclavo/.style  = {->, line width=1.1pt, verde!45!black},
  nota/.style     = {font=\scriptsize, align=center, inner sep=2pt, rojo!75!black},
  cond/.style     = {font=\scriptsize, align=center, inner sep=2pt},
}
% nombre del estado y lo que vale en él
\newcommand{\est}[2]{{\small\bfseries #1}\\[2pt]#2}

\begin{document}
\begin{tikzpicture}[fondo]

% ============================================================== estados ===
\node[estado] (rep) at (0,0)    {\est{REPOSO}{\texttt{wen} = \texttt{ren} = 0}};
\node[estado] (ped) at (4.8,0)  {\est{PEDIDO}{un ciclo: el estrobo\\ escribe o lanza\\ la lectura}};
\node[estado] (esp) at (9.6,0)  {\est{ESPERA}{\texttt{wen}/\texttt{ren} altos\\ \texttt{addr}, \texttt{wdata}\\ quietos}};
\node[estado, fill=verde!25] (ack) at (14.4,0) {\est{ACK}{\texttt{ack} = 1\\ \texttt{rdata} registrado}};

% ========================================================== transiciones ===
\draw[maestro] (rep) -- node[cond, above] {el CDC levanta\\ \texttt{wen} o \texttt{ren}} (ped);
\draw[maestro] (ped) -- node[cond, above] {ciclo\\ siguiente} (esp);
\draw[esclavo] (esp) -- node[cond, above] {\texttt{en\_sr[3]}} node[cond, below] {latencia fija,\\ toda dirección} (ack);
\path (esp) edge[maestro, loop above, looseness=5, min distance=8mm]
  node[cond, above] {sin \texttt{ack}: sigue} (esp);
\draw[maestro] (ack.north) -- ++(0,1.95) -| node[cond, above, pos=0.25] {el CDC ve el \texttt{ack} y baja \texttt{wen}/\texttt{ren}} (rep.north);

% si el ack no llega nunca, no hay transición: el esclavo se queda en ESPERA
\draw[rojo!75!black, densely dashed, line width=0.6pt] (esp.south) -- ++(0,-0.55) node[nota, below]
  {si el \texttt{ack} no llega, de acá no se sale:\\ el \emph{timeout} de \texttt{axi4\_slave} (32 ciclos)\\
   libera al ARM, no al slot};

% ============================================================== leyenda ===
\begin{scope}[shift={(-1.3,-2.95)}, every node/.style={font=\scriptsize, anchor=west}]
  \draw[maestro] (0,0) -- (0.7,0);   \node at (0.8,0)  {lo hace el maestro (\texttt{sys\_bus\_cdc})};
  \draw[esclavo] (5.3,0) -- (6.0,0); \node at (6.1,0)  {obligación del esclavo};
\end{scope}

\end{tikzpicture}
\end{document}
