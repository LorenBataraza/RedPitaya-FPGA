% El deep memory del osciloscopio stock (rp_axi_sm), dibujado como generador,
% buffer y consumidor, igual que docs/figuras/buffering_generador_consumidor.tex.
% Las muestras van a DDR por HP y no cruzan el bus: en DDR hay lugar para muchas
% ventanas. Pero dónde cayó el trigger se anota en UN registro (WP_TRIG, uno por
% canal), y la máquina es la misma de rp_bram_sm: tras el post-trigger la
% entrada queda congelada hasta que el PS lee la marca por el bus y re-arma.
% Abajo, el mismo tren de nueve eventos que en (a) de la figura de buffering:
% K = 1 otra vez. Es un esquema, no una medición.
% Fuente: rtl/classic/rp_axi_sm.v — docs/osc/sistema_adquisicion_original.md §5.
%
% Compilar:   pdflatex deep_memory_una_marca.tex
%             pdftocairo -svg deep_memory_una_marca.pdf deep_memory_una_marca.svg
%

% Línea de tiempo: t en unidades de τ, x = \xo + \ux·t (la misma que la figura
% de buffering, para que los trenes se puedan comparar).
\def\xo{2.1} \def\ux{0.85}
\newcommand{\tx}[1]{{\xo+\ux*(#1)}}
\newcommand{\arribo}[2]{\draw[azul!55!black, line width=1.2pt] (\tx{#1},#2) -- ++(0,0.45);}
\newcommand{\perdido}[2]{\draw[rojo!75!black, line width=1.2pt] (\tx{#1},#2) -- ++(0,0.26);}
\newcommand{\servicio}[3]{%
  \fill[black!22] (\tx{#1},#3) rectangle (\tx{#2},{#3+0.32});
  \draw[white, line width=1pt] (\tx{#1},#3) -- ++(0,0.32);}
\newcommand{\congelada}[3]{\fill[rojo!28] (\tx{#1},#3) rectangle (\tx{#2},{#3+0.2});}
\newcommand{\carril}[2]{\node[font=\scriptsize, black!75, anchor=east, inner sep=1pt] at (\xo-0.15,#1) {#2};}
\newcommand{\eje}[1]{%
  \draw[->, black!55, line width=0.5pt] (\xo,#1) -- (\tx{12.3},#1);
  \node[font=\scriptsize, black!60, anchor=west, inner sep=1pt] at (\tx{12.3},#1) {tiempo};
  \foreach \k in {0,...,12} \draw[black!55, line width=0.4pt] (\tx{\k},#1) -- ++(0,-0.06);}

\tikzset{
  rearma/.style = {->, rojo!65!black, line width=0.9pt, densely dashed},
  apunta/.style = {->, azul!60!black, line width=0.6pt, densely dotted},
  pierde/.style = {font=\scriptsize, rojo!75!black, inner sep=1pt},
  rotb/.style   = {font=\scriptsize, inner sep=1.5pt, fill=white, text=black},
}
