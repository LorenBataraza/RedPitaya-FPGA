% Una medición del MCA, vista desde el PS: el contrato PS <-> mca.
% Fuente: modulos/mca/rtl/mca_top.sv (mapa "MCA2", docs/mca/register_map_mca.md)
% y software/libmca/src/mca.c (mca_configure; mca_clear, mca_start, mca_stop,
% esperar_vaciado, leer_resultado: lo que hace mca_read_acquire, en ese orden).
% La lista de registros de la configuración va en el epígrafe: en la flecha no entra.
%
% La diferencia con el osciloscopio es lo que pasa en la franja verde: cada
% evento se procesa y se suma al histograma en la PL, sin el PS. El PS habla con
% el módulo al principio y al final de la medición, no por evento.
%
% Las reglas de cada paso (qué pasa si no se respeta) van en el epígrafe,
% secuencia.json, no en la figura. El trigger es una entrada del módulo: de
% dónde sale es asunto del multitrigger, no de este contrato.
%
% Compilar:   pdflatex mca_secuencia.tex
%             pdftocairo -svg mca_secuencia.pdf mca_secuencia.svg
%
\documentclass[border=0pt,tikz]{standalone}
\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage[none]{hyphenat}
\input{../../../figuras/estilo_tikz.tex}

\begin{document}
\begin{tikzpicture}[fondo]
\def\xm{0} \def\xps{10.0}
\def\yfin{-12.45}

\node[actor=verde!55] at (\xm,0)  {\textbf{mca}\\ registros e histogramas};
\node[actor=grisr]    at (\xps,0) {\textbf{PS}\\ software};
\foreach \x in {\xm,\xps} \draw[vida] (\x,-0.43) -- (\x,\yfin);

% ---- 1: configurar, una vez
\draw[gp] (\xps,-1.3) -- node[rot] {configurar: \texttt{CTRL.chan\_sel} con \texttt{run} = 0, y los \texttt{cfg\_*}} (\xm+0.16,-1.3);

% ---- el marco de una medición
\fill[marcolazo] (-4.7,-2.2) rectangle (14.4,-12.25);
\draw[marcolazo, fill=none] (-4.7,-2.2) rectangle (14.4,-12.25);
\node[rotlazo, anchor=north west] at (-4.7,-2.2) {una medición};
\foreach \x in {\xm,\xps} \draw[vida] (\x,-2.2) -- (\x,-12.25);

% estados del módulo, los que el PS ve en CTRL y STATUS
\franja{\xm}{-2.85}{-4.1}{amarillo}{borrando:\\ \texttt{clear\_busy} = 1}
\franja{\xm}{-4.1}{-5.0}{grisr}{parado, en cero}
\franja{\xm}{-5.0}{-8.45}{verde!55}{}
\node[rotfranja] at (\xm-0.3,-7.45) {midiendo: cada trigger abre\\ una ranura y, al cerrar,\\
  suma 1 a un bin y a los\\ contadores, sin el PS};
\franja{\xm}{-8.45}{-9.55}{amarillo}{vaciando: las ranuras\\ abiertas cierran}
\franja{\xm}{-9.55}{-12.25}{grisr}{parado:\\ el histograma, quieto}

% ---- 2: borrar, y esperar el barrido
\draw[gp] (\xps,-2.85) -- node[rot] {escribir \texttt{CTRL.clear}} (\xm+0.16,-2.85);
\draw[gp] (\xps,-3.5) -- node[rot] {leer \texttt{STATUS}: \texttt{clear\_busy} = 1} (\xm+0.16,-3.5);
\draw[gp] (\xps,-4.35) -- node[rot] {leer \texttt{STATUS}: \texttt{clear\_busy} = 0} (\xm+0.16,-4.35);
\repite{\xps}{-3.5}{-4.35}{se repite\\ hasta \texttt{clear\_busy} = 0}

% ---- 3: correr; los eventos entran solos
\draw[gp] (\xps,-5.0) -- node[rot] {escribir \texttt{CTRL.run} = 1} (\xm+0.16,-5.0);
\draw[entrada] (-4.3,-5.55) -- node[rot, fill=azul!8, above=1pt, pos=0.45] {triggers}
  (\xm-0.16,-5.55);
\draw[entrada] (-4.3,-5.85) -- (\xm-0.16,-5.85);
\draw[entrada] (-4.3,-6.15) -- (\xm-0.16,-6.15);
\node[font=\scriptsize, black!60] at (-2.2,-6.45) {\dots};
\draw[gp, densely dashed] (\xps,-6.7) -- node[rot] {leer contadores y relojes: opcional,
  con la medición corriendo} (\xm+0.16,-6.7);
\propio[black!70]{\xps}{-7.6}{esperar el tiempo\\ de medición}

% ---- 4: parar, y esperar que las ranuras publiquen
\draw[gp] (\xps,-8.45) -- node[rot] {escribir \texttt{CTRL.run} = 0} (\xm+0.16,-8.45);
\draw[gp] (\xps,-9.05) -- node[rot] {leer \texttt{STATUS}: \texttt{busy} = 1} (\xm+0.16,-9.05);
\draw[gp] (\xps,-9.8) -- node[rot] {leer \texttt{STATUS}: \texttt{busy} = 0} (\xm+0.16,-9.8);
\repite{\xps}{-9.05}{-9.8}{se repite\\ hasta \texttt{busy} = 0}

% ---- 5: el resultado
\draw[gp] (\xm+0.16,-10.55) -- node[rot] {leer el espectro 1D (\texttt{0x10000})\\
  y el mapa 2D (\texttt{0x20000})} (\xps,-10.55);
\draw[gp] (\xm+0.16,-11.2) -- (\xps,-11.2);
\repite{\xps}{-10.55}{-11.2}{un bin por transacción:\\ 8192 del espectro\\ y 8192 del mapa 2D}
\draw[gp] (\xps,-11.8) -- node[rot] {leer contadores y relojes} (\xm+0.16,-11.8);

% ---- y de vuelta
\draw[->, line width=0.6pt, black!70] (\xps,-12.75) -- (\xps+0.9,-12.75)
  node[right, font=\scriptsize, black!70, align=left] {vuelve: \texttt{clear} y \texttt{run}, o sólo
  \texttt{run}\\ para acumular; la configuración queda};

% leyenda
\draw[gp] (-4.7,-13.5) -- ++(0.7,0);
\node[font=\scriptsize, anchor=west] at (-3.95,-13.5)
  {registro o apertura de histograma, por GP0};
\draw[gp, densely dashed] (-4.7,-13.95) -- ++(0.7,0);
\node[font=\scriptsize, anchor=west] at (-3.95,-13.95) {opcional};
\draw[entrada] (-4.7,-14.4) -- ++(0.7,0);
\node[font=\scriptsize, anchor=west] at (-3.95,-14.4)
  {entrada del módulo: no la maneja el PS};
\draw[->, black!80, line width=0.7pt, rounded corners=2pt] (-4.0,-14.72) -- (-3.85,-14.72) -- (-3.85,-14.98) -- (-4.0,-14.98);
\node[font=\scriptsize, anchor=west] at (-3.75,-14.85) {lo que el PS repite};
\end{tikzpicture}
\end{document}
