% Cómo se suman las pérdidas en una red de procesamiento feed-forward, en un
% solo diagrama y sin texto: la explicación va en
% perdida_transacciones_vs_cuentas_back_pressure.txt.
% a) Cadena lineal: la etapa 3 se frena (contrapresión) y su parte la descarta
%    la 2; la pérdida total es la suma de lo que descarta cada etapa.
% b) Ramas exclusivas, con los componentes de handshake de los circuitos
%    asíncronos: DEMUX (con su canal de control) y MERGE.
% c) Ramas no exclusivas: el fork (el punto) copia cada evento a las dos; el
%    nudo "?" es el criterio de supervivencia, y el Venn, los perdidos.
% Números de ejemplo, no una medición.
%
% Compilar:   pdflatex perdida_transacciones_vs_cuentas_back_pressure.tex
%             pdftocairo -svg perdida_transacciones_vs_cuentas_back_pressure.pdf perdida_transacciones_vs_cuentas_back_pressure.svg
%

\tikzset{

  descarte/.style = {->, rojo!75!black, line width=0.9pt},
  perdida/.style  = {font=\scriptsize, rojo!75!black, inner sep=1pt},
  contra/.style   = {->, black!70, line width=0.8pt},

}

% Componentes de handshake. #1 = nombre, #2 = x del centro (y = 0 en el scope).
% DEMUX: lado angosto (la entrada) a la izquierda, el ancho (las salidas) a la
% derecha, y el canal de control por abajo. MERGE: el espejo, sin control.
\newcommand{\demux}[2]{%
  \draw[line width=0.6pt, fill=grisr] (#2-0.3,0.45) -- (#2+0.3,1.0) --
    (#2+0.3,-1.0) -- (#2-0.3,-0.45) -- cycle;
  \node[font=\tiny\ttfamily, rotate=90] at (#2,0) {DEMUX};
  \coordinate (#1-in) at (#2-0.3,0);
  \coordinate (#1-a)  at (#2+0.3,0.7);
  \coordinate (#1-b)  at (#2+0.3,-0.7);
  \draw[ctrl] (#2,-1.45) node[below, font=\tiny\ttfamily, inner sep=1pt] {ctl} -- (#2,-0.73);}
\newcommand{\merge}[2]{%
  \draw[line width=0.6pt, fill=grisr] (#2-0.3,1.0) -- (#2+0.3,0.45) --
    (#2+0.3,-0.45) -- (#2-0.3,-1.0) -- cycle;
  \node[font=\tiny\ttfamily, rotate=90] at (#2,0) {MERGE};
  \coordinate (#1-a)   at (#2-0.3,0.7);
  \coordinate (#1-b)   at (#2-0.3,-0.7);
  \coordinate (#1-out) at (#2+0.3,0);}
