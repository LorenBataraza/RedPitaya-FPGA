% Mux de una decisión: trapecio con el lado ancho (las entradas) a la izquierda.
% #1 = nombre, #2 = centro, #3 = alto en cm. Deja tres coordenadas:
%   (#1-in)  centro del lado de entradas: las flechas llegan a (#1-in |- opcion.east)
%   (#1-out) la salida
%   (#1-sel) mitad del lado de abajo: por ahí entra el registro que elige
% El rótulo de adentro es opcional: \muxdec{m}{(4,0)}{3.6} y después
% \node[...] at (m) {...}.
\newcommand{\muxdec}[3]{%
  \path #2 coordinate (#1);
  \draw[line width=0.6pt, fill=grisr]
    ($(#1)+(-0.45,{#3/2})$) -- ($(#1)+(0.45,{#3/2-0.35})$) --
    ($(#1)+(0.45,{-#3/2+0.35})$) -- ($(#1)+(-0.45,{-#3/2})$) -- cycle;
  \coordinate (#1-in)  at ($(#1)+(-0.45,0)$);
  \coordinate (#1-out) at ($(#1)+(0.45,0)$);
  \coordinate (#1-sel) at ($(#1)+(0,{-#3/2+0.175})$);}

% El mismo mux rotado 90 grados, para un flujo de arriba hacia abajo: entradas
% por el lado ancho de arriba, salida abajo y el registro que elige por la
% izquierda. #1 = nombre, #2 = centro, #3 = ancho en cm. Deja (#1-in) centro del
% lado de entradas, (#1-out) la salida y (#1-sel) mitad del lado izquierdo.
\newcommand{\muxdecv}[3]{%
  \path #2 coordinate (#1);
  \draw[line width=0.6pt, fill=grisr]
    ($(#1)+({-#3/2},0.45)$) -- ($(#1)+({#3/2},0.45)$) --
    ($(#1)+({#3/2-0.35},-0.45)$) -- ($(#1)+({-#3/2+0.35},-0.45)$) -- cycle;
  \coordinate (#1-in)  at ($(#1)+(0,0.45)$);
  \coordinate (#1-out) at ($(#1)+(0,-0.45)$);
  \coordinate (#1-sel) at ($(#1)+({-#3/2+0.175},0)$);}

% Leyenda de colores + nota. #1 = posición, #2 = color/texto, #3 = nota.
% Con la nota vacía ({}) sólo dibuja los colores: los panoramas no llevan
% epígrafe adentro, va en su .txt y en el doc que los incluye.
\newcommand{\leyenda}[3]{%
  \begin{scope}[shift={(#1)}, every node/.style={font=\scriptsize}]
    \coordinate (t0) at (-0.5,0);
    \foreach \c/\t [count=\i] in {#2} {
      \pgfmathtruncatemacro\j{\i-1}
      \node[draw, fill=\c, minimum width=4mm, minimum height=3mm, inner sep=0pt,
            right=5mm of t\j.east, anchor=west] (l\i) {};
      \node[right=1.5mm of l\i] (t\i) {\t};
    }
    \if\relax\detokenize{#3}\relax\else
      \node[anchor=north west, black!70, text width=19cm, align=left]
        at (-0.05,-0.3) {#3};
    \fi
  \end{scope}}

% ------------------------------------------------------------- secuencias ---
% DIAGRAMAS DE SECUENCIA: el contrato entre el PS y UN módulo
% (docs/<módulo>/figuras/secuencia/; el patrón salió de
% event_ring/figuras/ring_lote.tex). Una línea de vida por actor y el tiempo
% hacia abajo, sin escala: lo que fija la figura es el orden.
%
%   - Sólo el PS y el módulo. Lo que llega de otro módulo (el trigger, por
%     ejemplo) entra como una flecha \entrada desde el borde: es una entrada
%     del módulo, no un actor. Lo que sale hacia otro módulo (el disparo del
%     multitrigger) va al borde con salida, del mismo color.
%   - El módulo va con el color de su región (azul osc, amarillo mt, verde
%     mca, naranja event_ring); el PS, en gris.
%   - El color de la flecha dice por dónde viaja: rojo = registro o apertura
%     por GP0, naranja = DDR por HP.
%   - Los estados del módulo que el PS puede observar van como franjas sobre
%     su línea de vida (\franja), con el rótulo a la izquierda.
%   - La figura muestra el orden y nada más: qué pasa si un paso no se respeta
%     va en el epígrafe (secuencia.json), no en una columna de texto al lado.
\tikzset{

  hp/.style      = {->, naranja!60!black, line width=1.1pt},
  entrada/.style = {->, amarillo!35!black, line width=1.1pt},
  salida/.style  = {entrada},
  rot/.style     = {font=\scriptsize, fill=white, text=black, inner sep=1.5pt,

  % es un bloque sino un tramo de la secuencia
  marcolazo/.style = {fill=azul!8, draw=azul!60!black, line width=0.6pt,
                      rounded corners=3pt, dash pattern=on 3pt off 2pt},
  rotlazo/.style = {font=\scriptsize\bfseries, azul!40!black, inner sep=3pt},
  rotfranja/.style = {font=\scriptsize, black!75, anchor=east, align=right,
                      inner sep=1.5pt},
}

% \franja{x}{y0}{y1}{color}{rótulo}: estado del módulo sobre su línea de vida,
% de y0 a y1 (y0 arriba). El rótulo va a la izquierda, a media altura.
\newcommand{\franja}[5]{%
  \filldraw[fill=#4, draw=black!60, line width=0.5pt]
    ({#1-0.16},#2) rectangle ({#1+0.16},#3);
  \node[rotfranja] at ({#1-0.3},{(#2+#3)/2}) {#5};}

% \repite{x}{y0}{y1}{rótulo}: lo que el PS repite. Una flecha de vuelta a la
% derecha de su línea de vida, de y1 (la última vez) a y0 (la primera), con el
% rótulo al lado: hasta cuándo, o cuántas veces.
\newcommand{\repite}[4]{%
  \draw[->, black!80, line width=0.7pt, rounded corners=3pt]
    ({#1+0.1},#3) -- ({#1+0.5},#3) -- ({#1+0.5},#2) -- ({#1+0.1},#2);
  \node[font=\scriptsize, black!80, anchor=west, align=left, inner sep=1.5pt]
    at ({#1+0.6},{(#2+#3)/2}) {#4};}

% \propio[color]{x}{y}{rótulo}: lo que el PS hace solo, sin cruzar a la PL
% (leer su DDR, esperar). Una flecha que sale de su línea de vida y vuelve; el
% color por defecto es el de la DDR.
\newcommand{\propio}[4][naranja!60!black]{%
  \draw[->, #1, line width=1.1pt, rounded corners=3pt]
    ({#2+0.1},{#3+0.15}) -- ({#2+0.5},{#3+0.15}) -- ({#2+0.5},{#3-0.15}) -- ({#2+0.1},{#3-0.15});
  \node[font=\scriptsize, black, anchor=west, align=left, inner sep=1.5pt]
    at ({#2+0.6},#3) {#4};}


% ------------------------------------------------- símbolos de operación ---
% DATAPATHS A NIVEL DE OPERACIÓN (docs/gen_eventos/figuras/): una operación que
% tiene símbolo se dibuja con el símbolo, no con un bloque de texto. Tampoco se

%   puerta o   compuerta OR
