% Dónde se frena y dónde se descarta cuando algo no alcanza: de la captura a la
% DDR. Con las convenciones de docs/figuras/etapas_vs_descartes.tex y
% perdida_transacciones_vs_cuentas_back_pressure.tex: la etiqueta dice si la
% etapa descarta o bloquea, la flecha roja es el descarte (va a su contador) y
% las de vuelta, la contrapresión. Sólo la captura descarta, porque su entrada
% no tiene ready.
% Los números son las filas de la tabla de logica_de_captura_y_axi.md §5.6. Los
% créditos en sí (wr_slot, rd_slot sobre el ring) están en ring_creditos.tex. El
% crédito vuelve por GP0 (RD_SLOT, que escribe el PS), no desde la DDR.
% El epígrafe no va en la figura: está en ring_freno_descarte.txt y en el doc
% que la incluye.
% Fuente: modulos/event_ring/rtl/event_ring_top.sv (credit_ok, wr_rdy),
% event_slot_writer.sv y event_window_capture.sv.
%
% Compilar:   pdflatex ring_freno_descarte.tex
%             pdftocairo -svg ring_freno_descarte.pdf ring_freno_descarte.svg
%

}

% contrapresión: hacia atrás, sin perder nada

\node[comp, text width=1.6cm, minimum height=1.05cm] (ps) at (16,-2.55)

\draw[flujo, naranja!60!black] (ddr.south) -- node[sen, right, black, inner sep=2pt] {lee el lote} (ps.north);

% descarte: la captura pierde el evento entero y lo cuenta, por dos causas que
% salen de lugares distintos. Arriba, la propia: el trigger cae con la captura

\draw[descarte] ([xshift=-5mm]cap.south) -- ++(0,-0.75) coordinate (df);
\node[contador] (cf) at ($(df)+(0,-0.24)$) {DROP\_FULL};
\node[marca, fill=rojo!75!black, anchor=east] at ([xshift=-3pt]cf.west) {3};
\node[causa, anchor=north] at ([yshift=-1pt]cf.south) {la captura está libre, pero\\ aguas abajo no entra la ventana};
