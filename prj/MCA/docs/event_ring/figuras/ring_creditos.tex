% \anillo{centro x}{rd_slot}{wr_slot}: 12 slots, en sentido horario desde arriba
\newcommand{\anillo}[3]{%
  \begin{scope}[shift={(#1,0)}]
    \pgfmathtruncatemacro\ocup{#3-#2}
    \foreach \i in {0,...,11} {
      \pgfmathsetmacro\aa{90-\i*30}
      \pgfmathsetmacro\ab{90-(\i+1)*30}
      \pgfmathtruncatemacro\rel{mod(\i-mod(#2,12)+12,12)}
      \pgfmathtruncatemacro\lleno{\rel<\ocup ? 1 : 0}
      \ifnum\lleno=1 \def\colr{naranja!80}\else\def\colr{white}\fi
      \draw[fill=\colr, line width=0.5pt]
        (\aa:1.05) arc[start angle=\aa, end angle=\ab, radius=1.05] --
        (\ab:1.62) arc[start angle=\ab, end angle=\aa, radius=1.62] -- cycle;
      \pgfmathsetmacro\am{90-(\i+0.5)*30}
      \node[font=\fontsize{5.4}{6.4}\selectfont, inner sep=0pt] at (\am:1.335) {\i};
    }
    % punteros: wr por fuera, rd por dentro
    \pgfmathtruncatemacro\iw{mod(#3,12)}
    \pgfmathtruncatemacro\ir{mod(#2,12)}
    \pgfmathsetmacro\aw{90-(\iw+0.5)*30}
    \pgfmathsetmacro\ar{90-(\ir+0.5)*30}
    \draw[->, line width=0.9pt] (\aw:2.05) -- (\aw:1.65);
    \node[font=\scriptsize\ttfamily\bfseries, inner sep=1pt] at (\aw:2.25) {wr};
    \draw[->, line width=0.9pt, black!70] (\ar:0.55) -- (\ar:1.02);
    \node[font=\scriptsize\ttfamily\bfseries, inner sep=1pt, black!70] at (\ar:0.38) {rd};
  \end{scope}}
\newcommand{\pie}[6]{% x, título, wr, rd, ocupados, nota
  \node[font=\scriptsize\bfseries, anchor=south, inner sep=0pt] at (#1,2.62) {#2};
  \node[font=\scriptsize, anchor=north, align=center, inner sep=0pt, text width=4.3cm] at (#1,-2.62)
    {\texttt{wr\_slot} = #3,\; \texttt{rd\_slot} = #4\\ \texttt{wr $-$ rd} = #5\\[1pt] #6};}
