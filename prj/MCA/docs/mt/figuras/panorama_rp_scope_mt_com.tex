% Un dsp_tap_mux apuntando hacia abajo: #1 = x del centro, #2 = y del borde
% ancho, #3 = consumidor. Las cuatro entradas cuelgan del bus de taps (\yb).
\newcommand{\muxabajo}[3]{%
  \draw[line width=0.6pt, fill=grisr] (#1-0.8,#2) -- (#1+0.8,#2)
    -- (#1+0.4,#2-0.8) -- (#1-0.4,#2-0.8) -- cycle;
  \foreach \k in {0,...,3} {
    \draw[line width=0.8pt] (#1-0.6+0.4*\k,\yb) -- (#1-0.6+0.4*\k,#2);
    \node[font=\tiny, inner sep=0pt] at (#1-0.6+0.4*\k,#2-0.13) {\k};
  }
  \node[font=\scriptsize\bfseries, inner sep=0pt] at (#1,#2-0.5) {#3};}
