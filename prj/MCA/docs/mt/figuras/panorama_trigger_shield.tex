% Panorama de TRIGGER_SHIELD en el estilo de panorama_mt.tex: el escudo,
% registro por registro. Cada fila es un registro del RTL con la lógica que da
% su valor siguiente:
%   src_fire  = |(trig_event & src)                         (combinacional)
%   cnt       <- src_fire ? dur : (active ? cnt-1 : cnt)
%   active    <- src_fire ? (dur != 0) : (active & !vence)
%   clr_q     <- dst[N-1:0] & (src_fire ? (dur == 0) : vence)
%   mca_veto  <- dst[N] & (src_fire | active)
% con vence = active & (cnt <= 1): el holdoff termina. Mientras corre, cnt >= 1,
% así que cnt-1 es el «cnt <= 0» del RTL cuando vence.
% Las ramas del RTL (if src_fire / else if active) son los muxes en cascada.
% Fuente: modulos/mt/rtl/trigger_shield.sv. Estilo: docs/figuras/estilo_tikz.tex.
%
% EL COLOR ES LA FUNCIÓN, como en panorama_mt.tex: amarillo selección (de qué
% fuentes se dispara el escudo), naranja escudo, gris registros. Las operaciones
% van con su símbolo. Nombres de red sin cable (src_fire, active, vence,
% dst[N]): la señal del mismo nombre, que se dibuja una sola vez donde nace.
% El epígrafe no va en la figura: está en panorama_trigger_shield.txt y en el
% doc que la incluye.
%
% Compilar:   pdflatex panorama_trigger_shield.tex
%             pdftocairo -svg panorama_trigger_shield.pdf panorama_trigger_shield.svg
%
\documentclass[border=0pt,tikz]{standalone}

\tikzset{
  ysel/.style  = {puerta y=#1, fill=amarillo},
  ysel/.default = nn,
  osel/.style  = {puerta o, fill=amarillo},
  yesc/.style  = {puerta y=#1, fill=naranja!70},
  yesc/.default = nn,

  reg/.style   = {registro=#1, minimum height=0.7cm, font=\scriptsize\ttfamily},
  reg/.default = 0.8cm,
  chico/.style = {font=\scriptsize\ttfamily, inner sep=1pt},
  red/.style   = {ctrl},
}
% un nombre de red que entra por abajo a #1 (un punto), con el nombre #2
\newcommand{\redAbajo}[2]{\draw[red] ($(#1)+(0,-0.4)$) node[below, chico, black!80] {#2} -- (#1);}
% ... y por arriba
\newcommand{\redArriba}[2]{\draw[red] ($(#1)+(0,0.4)$) node[above, chico, black!80] {#2} -- (#1);}
% ... y por la izquierda
\newcommand{\redIzq}[2]{\draw[red] ($(#1)+(-0.45,0)$) node[left, chico, black!80] {#2} -- (#1);}
% las etiquetas de las entradas de un mux: #1 = mux, #2 = y de la de arriba,
% #3 = su rótulo, #4 = y de la de abajo, #5 = su rótulo
\newcommand{\muxio}[5]{%
  \node[font=\tiny] at ($(#1-in |- 0,#2)+(0.13,0)$) {#3};
  \node[font=\tiny] at ($(#1-in |- 0,#4)+(0.13,0)$) {#5};}
% la mitad del lado de arriba de un mux de alto #2: (#1-top)
\newcommand{\selArriba}[2]{\path ($(#1)+(0,{#2/2-0.175})$) coordinate (#1-top);}
