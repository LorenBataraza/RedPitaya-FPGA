% Panorama de TRIGGER_SHIELD: el escudo por dentro. Un solo contador de holdoff
% que carga `dur` cuando dispara alguna fuente habilitada; al expirar emite un
% pulso de re-arm hacia los canales de destino, y mientras corre da el veto al
% MCA. El borde grueso es el límite del módulo; lo que lo cruza son sus
% puertos, sólo con el nombre.

%

%             pdftocairo -svg panorama_trigger_shield.pdf panorama_trigger_shield.svg
%

% ================================================================ bloques ===
\node[logica, minimum height=1.6cm] (dis) at (2.6,0.0)

\node[logica, minimum height=1.9cm] (cnt) at (6.4,0.0)

\node[logica, minimum height=1.5cm] (pul) at (10.4,1.7)
  {\etq{pulso de re-arm}{\texttt{clr\_q <= dst[N-1:0]}\\ al expirar; con \texttt{dur=0},\\ al disparar \;(1 ciclo)}};

  {\etq{veto al MCA}{\texttt{dst[N] \& (src\_fire | active)}\\ registrado}};
\node[logica, text width=2.4cm, minimum height=1.5cm,
      double copy shadow={shadow xshift=2pt, shadow yshift=-2pt}] (mux) at (14.5,1.7)
