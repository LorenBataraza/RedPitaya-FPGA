% Panorama de MT_SLOPE_TRIG en el estilo de panorama_mt.tex: el disparo por
% pendiente de un canal, registro por registro. De arriba hacia abajo:
%   configuración  L y G saturados, dos registros, y lo que sale de ellos
%                  (direcciones de las líneas, tomas, G=0); restart cuando
%                  cambian, con en=0 o con reset
%   líneas         x --z^-L--> xa --z^-G--> xb --z^-L--> xc, cada una con una
%                  toma dinámica (dirección L-1 o G-1); con G=0 la del medio
%                  se puentea
%   tomas          una toma más vieja que fill lee 0 (historia en cero)
%   diferencia     d = (x - xa) - (xb - xc)
%   acumulador     s <= s + d
%   Schmitt        p: S con s >= thr, R con s < thr-hyst; n: S con s <= -thr,
%                  R con s > -thr+hyst. Mientras la ventana no está llena
%                  (p3_full = 0) queda en 1, sin flanco
%   flanco         slope = scht[0] & !scht[1], registrado
% Fuente: modulos/mt/rtl/mt_slope_trig.sv; teoría en docs/mt/trigger_pendiente.md.
% Estilo: docs/figuras/estilo_tikz.tex.
%
% EL COLOR ES LA FUNCIÓN, como en panorama_mt.tex: azul detección (todo el
% módulo), gris registros. Las operaciones van con su símbolo. Nombres de red sin
% cable (dv, restart, fill, full, L, G, L-1, ..., thr_p, p3_v, p3_full): la
% señal del mismo nombre, que se dibuja una sola vez donde nace.
% El epígrafe no va en la figura: está en panorama_mt_slope_trig.txt y en el
% doc que la incluye.
%
% Compilar:   pdflatex panorama_mt_slope_trig.tex
%             pdftocairo -svg panorama_mt_slope_trig.pdf panorama_mt_slope_trig.svg
%

  reg/.style   = {registro=#1, minimum height=0.6cm, font=\scriptsize\ttfamily},

                      -- ++(0,0.16);}},
  chico/.style = {font=\scriptsize\ttfamily, inner sep=1pt},
  rot/.style   = {font=\tiny, inner sep=0.5pt, black!80},
  red/.style   = {ctrl},
}
% un nombre de red que entra a #1, desde la izquierda, arriba o abajo
\newcommand{\redIzq}[2]{\draw[red] ($(#1)+(-0.45,0)$) node[left, chico, black!80] {#2} -- (#1);}
\newcommand{\redArriba}[2]{\draw[red] ($(#1)+(0,0.4)$) node[above, chico, black!80] {#2} -- (#1);}
\newcommand{\redAbajo}[2]{\draw[red] ($(#1)+(0,-0.4)$) node[below, chico, black!80] {#2} -- (#1);}
% el segundo operando de una resta, desde abajo, con el nombre a la izquierda
\newcommand{\redAbajoIzq}[2]{\draw[red] ($(#1)+(0,-0.32)$) node[left, chico, black!80] {#2} -- (#1);}
% una constante que entra a #1 desde arriba
\newcommand{\cteArriba}[2]{\draw[flujo] ($(#1.north)+(0,0.36)$) node[above, cte] {#2} -- (#1.north);}
% una salida con nombre: flecha corta y el nombre
\newcommand{\nace}[2]{\draw[flujo] (#1) -- ++(0.4,0) node[right, chico] {#2};}
